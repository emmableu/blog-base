暴力解
```python
from collections import Counter
class Solution:
    def hIndex(self, citations: List[int]) -> int:
        # [10,3,2,1,0] => 2 i.e, 10>=2, 3>=2, (2>=2)
        # [6,0] => 1 => i.e., 6>=1
        for i in range(len(citations), -1, -1):
            num_data = len([ele for ele in citations if ele >= i])
            if num_data >= i:
                return i
        return 0
```

注意：
1. `            num_data = len([ele for ele in citations if ele >= i])` 一定要有中括号
2. `        for i in range(len(citations), -1, -1):` 第二个-1 （第三个参数）一定要有

```python
from collections import Counter, defaultdict
class Solution:
    def hIndex(self, citations: List[int]) -> int:
        cnt = defaultdict(int)
        for ele in citations:
            ele = min(ele, len(citations))
            cnt[ele] += 1
        count_with_at_least_cur_citation = 0
        for i in range(len(citations), -1, -1):
            count_with_at_least_cur_citation += cnt[i]
            if count_with_at_least_cur_citation >= i:
                return i 
        return 0
```
