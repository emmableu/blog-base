```python
"""
      |   |
    | |   | |
  | | | | | |
| | | | | | |
0 1 2 3 4 5 6
0: is_up, nums[1]>nums[2]
1: is_up, nums[2]>nums[1]
2: is_up, nums[3]>nums[2]
3: is_down, nums[4]<nums[3]
4: is_up, nums[5]>nums[4]
5: is_down, nums[6]<nums[5]
goal: find the first is_down, after all of the is_up. i.e., 3
if saw is_up, go right, i.e., l = mid
if saw is_down, go left. i.e, r = mid
"""


class Solution:
    def findPeakElement(self, nums: List[int]) -> int:
        l, r = -1, len(nums) - 1 # len(nums) - 1 as we will not go to the last element
        while l + 1 < r:
            mid = (l + r) // 2
            if self.is_up(nums, mid):
                l = mid 
            elif self.is_down(nums, mid):
                r = mid
        return l + 1
    

    def is_up(self, nums, mid):
        return nums[mid + 1] > nums[mid]

    def is_down(self, nums, mid):
        return nums[mid + 1] < nums[mid]
```
